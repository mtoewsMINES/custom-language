<prog> -> 
    | <stmt> <prog>
    | "{" <prog> "}"
    | ""
<stmt> ->
    #variable declaration/assignment
    | <ptype> <ident> "->" <exp> ";"
    | <ident> "->" <exp> ";"
    | "if" <exp> <prog> "else" <prog>
<exp> ->
    | <bop>
<bop> ->
    | <term> <bop'>
<bop'> ->
    | "+" <term> <bop'>
    | "-" <term> <bop'>
    | "||" <term> <bop'>
    | ""
<term> ->
    | <factor> <term'>
<term'> ->
    | "*" -> <factor> <term'>
    | "/" -> <factor> <term'>
    | "&&" -> <factor> <term'>
    | ""
<factor> ->
    | "(" <exp> ")"
    | <ident>
    | int | string | bool
    | <uop>
<uop> ->
    | "!" <exp>


<ptype> -> "int" | "string" | "bool"
<ident> -> alphanumeric characters
